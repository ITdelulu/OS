#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <sys/wait.h>

int main()
{
    int n, a[20], i, j, temp;
    pid_t pid;

    printf("Enter number of integers: ");
    scanf("%d", &n);

    printf("Enter integers:\n");
    for(i = 0; i < n; i++)
        scanf("%d", &a[i]);

    pid = fork();

    if(pid < 0)
    {
        printf("Fork failed");
        return 1;
    }

    if(pid == 0)
    {
        // Child process: Ascending order
        for(i = 0; i < n-1; i++)
            for(j = 0; j < n-i-1; j++)
                if(a[j] > a[j+1])
                {
                    temp = a[j];
                    a[j] = a[j+1];
                    a[j+1] = temp;
                }

        printf("\nChild Process (Ascending): ");
        for(i = 0; i < n; i++)
            printf("%d ", a[i]);

        printf("\n");
        exit(0);
    }
    else
    {
        // Parent process: Descending order
        for(i = 0; i < n-1; i++)
            for(j = 0; j < n-i-1; j++)
                if(a[j] < a[j+1])
                {
                    temp = a[j];
                    a[j] = a[j+1];
                    a[j+1] = temp;
                }

        printf("\nParent Process (Descending): ");
        for(i = 0; i < n; i++)
            printf("%d ", a[i]);

        wait(NULL);
        printf("\nChild process completed.\n");
    }

    // Demonstration of Zombie Process
    if(getpid() == getppid()) {
        return 0;
    }

    pid = fork();

    if(pid == 0)
    {
        printf("\nZombie demo child exiting. PID: %d\n", getpid());
        exit(0);
    }
    else
    {
        printf("Parent sleeping for 10 seconds. Check child PID %d using ps.\n", pid);
        sleep(10);
        wait(NULL);
        printf("Zombie process collected by parent.\n");
    }

    // Demonstration of Orphan Process
    pid = fork();

    if(pid == 0)
    {
        sleep(3);
        printf("\nOrphan child running. New Parent PID: %d\n", getppid());
        exit(0);
    }
    else
    {
        printf("\nParent exiting. Child becomes orphan.\n");
        return 0;
    }
}
