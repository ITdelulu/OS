#include <stdio.h>

int main()
{
    int n, m;
    int a[10][10], max[10][10], need[10][10];
    int avail[10], finish[10] = {0}, safe[10];
    int count = 0;

    printf("Enter processes and resources: ");
    scanf("%d%d", &n, &m);

    printf("Enter Allocation Matrix:\n");
    for(int i=0; i<n; i++)
        for(int j=0; j<m; j++)
            scanf("%d", &a[i][j]);

    printf("Enter Maximum Matrix:\n");
    for(int i=0; i<n; i++)
        for(int j=0; j<m; j++)
            scanf("%d", &max[i][j]);

    printf("Enter Available Resources:\n");
    for(int j=0; j<m; j++)
        scanf("%d", &avail[j]);

    // Calculate Need
    for(int i=0; i<n; i++)
        for(int j=0; j<m; j++)
            need[i][j] = max[i][j] - a[i][j];

    while(count < n)
    {
        int found = 0;

        for(int i=0; i<n; i++)
        {
            if(!finish[i])
            {
                int j;
                for(j=0; j<m; j++)
                    if(need[i][j] > avail[j])
                        break;

                if(j == m)
                {
                    for(j=0; j<m; j++)
                        avail[j] += a[i][j];

                    safe[count++] = i;
                    finish[i] = 1;
                    found = 1;
                }
            }
        }

        if(!found)
            break;
    }

    if(count == n)
    {
        printf("\nSafe State\nSequence: ");
        for(int i=0; i<n; i++)
            printf("P%d ", safe[i]);
    }
    else
        printf("\nUnsafe State");

    return 0;
}
