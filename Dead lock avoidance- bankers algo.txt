
#include <stdio.h>

int main() {
    int alloc[5][3] = {
        {0,1,0}, {2,0,0}, {3,0,2},
        {2,1,1}, {0,0,2}
    };

    int max[5][3] = {
        {7,5,3}, {3,2,2}, {9,0,2},
        {2,2,2}, {4,3,3}
    };

    int avail[3] = {3,3,2};
    int need[5][3], finish[5] = {0};
    int i, j, k = 0, found, safe[5];

    // Calculate Need Matrix
    for(i = 0; i < 5; i++)
        for(j = 0; j < 3; j++)
            need[i][j] = max[i][j] - alloc[i][j];

    // Banker's Algorithm
    while(k < 5) {
        found = 0;

        for(i = 0; i < 5; i++) {
            if(!finish[i]) {
                for(j = 0; j < 3; j++)
                    if(need[i][j] > avail[j])
                        break;

                if(j == 3) {
                    for(j = 0; j < 3; j++)
                        avail[j] += alloc[i][j];

                    safe[k++] = i;
                    finish[i] = 1;
                    found = 1;
                }
            }
        }

        if(!found)
            break;
    }

    if(k == 5) {
        printf("Safe sequence: ");
        for(i = 0; i < 5; i++)
            printf("P%d ", safe[i]);
    }
    else {
        printf("Unsafe state");
    }

    printf("\n");
    return 0;
}
