
#include <stdio.h>
#include <stdlib.h>

int main() {
    int n, head, i, j, total = 0;
    int a[20], visited[20] = {0};

    printf("Enter number of requests: ");
    scanf("%d", &n);

    printf("Enter request queue:\n");
    for(i = 0; i < n; i++)
        scanf("%d", &a[i]);

    printf("Enter initial head position: ");
    scanf("%d", &head);

    printf("\nSeek Sequence: %d", head);

    // SSTF Algorithm
    for(i = 0; i < n; i++) {
        int min = 9999, pos = -1;

        for(j = 0; j < n; j++) {
            int diff = abs(head - a[j]);

            if(!visited[j] && diff < min) {
                min = diff;
                pos = j;
            }
        }

        visited[pos] = 1;
        total += abs(head - a[pos]);
        head = a[pos];

        printf(" -> %d", head);
    }

    printf("\nTotal head movement = %d\n", total);

    return 0;
}
